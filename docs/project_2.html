# Count words in each phrase.
hamlet_words <- hamlet |>
  mutate(
    word_count = str_count(
      dialogue,
      "\\b[[:alpha:]]+\\b"
    )
  )

# Add words per character, per act. 
act_words <- hamlet_words |>
  group_by(act, character) |>
  summarise(
    total_words = sum(word_count),
    .groups = "drop"
  )

Plot (Stacked Bar Chart)

Show Code 
act_plot <- ggplot(
  act_words,
  aes(
    x = act,
    y = total_words,
    fill = fct_reorder(character, -total_words, .fun = sum),
    text = paste0(
      "Act: ", act,
      "<br>Character: ", character,
      "<br>Words: ", total_words
    )
  )
) +
  geom_col() +
  labs(
    title = "Who Speaks the Most in Each Act of Hamlet?",
    x = "Act",
    y = "Total Words Spoken",
    fill = "Character"
  ) +
  theme_minimal()

ggplotly(act_plot, tooltip = "text")

### Question 1: Who speaks the most in each Act?
```{r}
# Count words in each phrase.
hamlet_words <- hamlet |>
  mutate(
```
